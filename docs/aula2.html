Vetores

Numéricos

# criar vetores numéricos

num <- c(10, 2.5, 4, esc)
num2 <- c(3, 4, 2, 2)
sort(num)
[1]  2.5  4.0 10.0 10.0
typeof(num)
[1] "double"
# operações com vetores e escalares

num + esc
[1] 20.0 12.5 14.0 20.0
num / esc
[1] 1.00 0.25 0.40 1.00
num * esc
[1] 100  25  40 100
num - esc
[1]  0.0 -7.5 -6.0  0.0
num ^ esc
[1] 1.000000e+10 9.536743e+03 1.048576e+06 1.000000e+10
# operação pairwise/element-wise

num + num2
[1] 13.0  6.5  6.0 12.0
num - num2
[1]  7.0 -1.5  2.0  8.0
num * num2
[1] 30 10  8 20
num / num2
[1] 3.333333 0.625000 2.000000 5.000000
# reciclagem (operação reciclada)

num3 <- c(num, num2)
num * num3
[1] 100.00   6.25  16.00 100.00  30.00  10.00   8.00  20.00

Caracteres/fatores/lógicos

char <- c("t1", "t2", "t3")
typeof(char)
[1] "character"
char2 <- c("pequeno", "médio", "grande", "gigante")
sort(char2)
[1] "gigante" "grande"  "médio"   "pequeno"
# fator

factor(char2, levels = char2)
[1] pequeno médio   grande  gigante
Levels: pequeno médio grande gigante
# fator numérico
factor(num)
[1] 10  2.5 4   10 
Levels: 2.5 4 10

Estruturas de dados

Data frames

alunos <- 
data.frame(
  nome = c("tiago", "gabriela", "jamille", "ana", "richard", "cleibi"),
  altura = c(191, 162, 169, 160, 190, 188),
  cromossomo = factor(x = c("xy", "xx", "xx", "xx", "xy", "xy"), levels = c("xy", "xx"))
)

Matrizes

# matriz com variável normal (média 30, dp = 4)

set.seed(1)
mat <- 
matrix(data = rnorm(n = 16, mean = 30, sd = 4), 
       nrow = 4, 
       ncol = 4)

# seleção de elementos

alunos[4,2]
[1] 160
# seleção de linhas

alunos[3, ]
     nome altura cromossomo
3 jamille    169         xx
# seleção de colunas

alunos[, 2]
[1] 191 162 169 160 190 188

Listas

lista <- 
  list(
    dataframe = alunos,
    matriz = mat,
    escalar = esc,
    numerico = num,
    fator = factor(char), 
    lista = list(df = alunos, mat = mat)
  )
